  (() => {
    const avisos = document.querySelectorAll('.aviso');

      const fechar = document.createElement('button');
      fechar.type = 'button';
      fechar.className = 'aviso__fechar';
      fechar.setAttribute('aria-label', 'Fechar aviso');

      });

      aviso.append(fechar);
    });
