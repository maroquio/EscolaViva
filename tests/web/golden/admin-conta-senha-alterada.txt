  <script>
  (() => {
    const avisos = document.querySelectorAll('.notice');
    if (avisos.length === 0) return;

    const conteudo = document.getElementById('content');

    avisos.forEach((aviso) => {
      const fechar = document.createElement('button');
      fechar.type = 'button';
      fechar.className = 'notice__close';
      fechar.setAttribute('aria-label', 'Fechar aviso');
      fechar.textContent = '×';

      fechar.addEventListener('click', () => {
        const tinhaFoco = document.activeElement === fechar;
        aviso.remove();
        if (tinhaFoco && conteudo !== null) conteudo.focus({ preventScroll: true });
      });

      aviso.append(fechar);
    });
  })();
</script>
