<script>
  (() => {
    const botao = document.getElementById('select-all');
    const caixas = Array.from(document.querySelectorAll('input[name="guardians[]"]'));
    const alcanceSelecionados = document.getElementById('audience-selected');
    if (botao === null || caixas.length === 0) return;

    const todasMarcadas = () => caixas.every((caixa) => caixa.checked);
    const atualizarRotulo = () => {
      botao.textContent = todasMarcadas() ? 'Desmarcar todos' : 'Selecionar todos';
    };

    botao.addEventListener('click', () => {
      const marcar = !todasMarcadas();
      caixas.forEach((caixa) => { caixa.checked = marcar; });
      if (marcar && alcanceSelecionados !== null) alcanceSelecionados.checked = true;
      atualizarRotulo();
    });

    caixas.forEach((caixa) => caixa.addEventListener('change', atualizarRotulo));

    atualizarRotulo();
    botao.hidden = false;
  })();
</script>

  <script>
  (() => {
    const avisos = document.querySelectorAll('.aviso');
    if (avisos.length === 0) return;

    const conteudo = document.getElementById('content');

    avisos.forEach((aviso) => {
      const fechar = document.createElement('button');
      fechar.type = 'button';
      fechar.className = 'aviso__fechar';
      fechar.setAttribute('aria-label', 'Fechar aviso');
      fechar.textContent = '×';

      fechar.addEventListener('click', () => {
        const tinhaFoco = document.activeElement === fechar;
        aviso.remove();
        if (tinhaFoco && conteudo !== null) conteudo.focus({ preventScroll: true });
      });

      aviso.append(fechar);
    });
  })();
</script>
