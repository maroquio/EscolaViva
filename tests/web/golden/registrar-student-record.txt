  (() => {
    const notices = document.querySelectorAll('.notice');
    if (notices.length === 0) return;

    const content = document.getElementById('content');

    notices.forEach((notice) => {
      const closeButton = document.createElement('button');
      closeButton.type = 'button';
      closeButton.className = 'notice__close';
      closeButton.setAttribute('aria-label', 'Fechar aviso');
      closeButton.textContent = '×';

      closeButton.addEventListener('click', () => {
        const hadFocus = document.activeElement === closeButton;
        notice.remove();
        if (hadFocus && content !== null) content.focus({ preventScroll: true });
      });

      notice.append(closeButton);
    });
  })();
</script>
