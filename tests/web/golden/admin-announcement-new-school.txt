<script>
  (() => {
    const button = document.getElementById('select-all');
    const checkboxes = Array.from(document.querySelectorAll('input[name="guardians[]"]'));
    const selectedAudience = document.getElementById('audience-selected');
    if (button === null || checkboxes.length === 0) return;

    const allChecked = () => checkboxes.every((checkbox) => checkbox.checked);
    const updateLabel = () => {
      button.textContent = allChecked() ? 'Desmarcar todos' : 'Selecionar todos';
    };

    button.addEventListener('click', () => {
      const shouldCheck = !allChecked();
      checkboxes.forEach((checkbox) => { checkbox.checked = shouldCheck; });
      if (shouldCheck && selectedAudience !== null) selectedAudience.checked = true;
      updateLabel();
    });

    checkboxes.forEach((checkbox) => checkbox.addEventListener('change', updateLabel));

    updateLabel();
    button.hidden = false;
  })();
</script>

  <script>
  (() => {
    const notices = document.querySelectorAll('.notice');
    if (notices.length === 0) return;

    const content = document.getElementById('content');

    notices.forEach((notice) => {
      const closeButton = document.createElement('button');
      closeButton.type = 'button';
      closeButton.className = 'notice__close';
      closeButton.setAttribute('aria-label', 'Fechar aviso');
      closeButton.textContent = '×';

      closeButton.addEventListener('click', () => {
        const hadFocus = document.activeElement === closeButton;
        notice.remove();
        if (hadFocus && content !== null) content.focus({ preventScroll: true });
      });

      notice.append(closeButton);
    });
  })();
</script>
